import pandas as pd
import numpy as np
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler

# Simulated cleaned dataset with 8021 data points
np.random.seed(0)
df_clean = pd.DataFrame({
    'Critic_Score': np.random.randint(50, 100, size=8021),
    'User_Score': np.random.uniform(5.0, 10.0, size=8021),
    'Global_Sales': np.random.uniform(0.1, 10.0, size=8021),
    'Rating': np.random.choice(['T', 'E', 'M', 'E10+'], size=8021)
})

# Step 1: Select features
features = ['Critic_Score', 'User_Score', 'Global_Sales']
X = df_clean[features]

# Step 2: Normalize features
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Step 3: Use custom seed points for initial centroids
seed_indices = [0, 2000, 4000, 6000]
initial_centroids = X_scaled[seed_indices]

# Step 4: KMeans clustering
kmeans = KMeans(n_clusters=4, init=initial_centroids, n_init=1, random_state=42)
df_clean['Cluster'] = kmeans.fit_predict(X_scaled)

# Step 5: Confusion matrix (counts only)
count_matrix = pd.crosstab(df_clean['Rating'], df_clean['Cluster'], margins=True, margins_name='Total')

# Step 6: Display results
print(f"Data shape: {df_clean.shape}")
print("\nRating vs Cluster Distribution (counts only):\n")
print(count_matrix)

import pandas as pd
import numpy as np
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
from IPython.display import display

# Simulated video game dataset with 8021 data points
np.random.seed(0)
df_clean = pd.DataFrame({
    'Critic_Score': np.random.randint(50, 100, size=8021),
    'User_Score': np.random.uniform(5.0, 10.0, size=8021),
    'Global_Sales': np.random.uniform(0.1, 10.0, size=8021),
    'Rating': np.random.choice(['T', 'E', 'M', 'E10+'], size=8021)
})

# Features for clustering
features = ['Critic_Score', 'User_Score', 'Global_Sales']
X = df_clean[features]

# Standardize features
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Define 10 different sets of initial seed indices (for 4 clusters)
seed_sets = [
    [0, 2000, 4000, 6000],
    [1, 2001, 4001, 6001],
    [2, 2002, 4002, 6002],
    [3, 2003, 4003, 6003],
    [4, 2004, 4004, 6004],
    [5, 2005, 4005, 6005],
    [6, 2006, 4006, 6006],
    [7, 2007, 4007, 6007],
    [8, 2008, 4008, 6008],
    [9, 2009, 4009, 6009]
]

# Iterate through each seed set
for i, seed_indices in enumerate(seed_sets, start=1):
    # Create a copy of the dataframe for this run
    df_temp = df_clean.copy()
    
    # Get initial centroids from the seed indices
    initial_centroids = X_scaled[seed_indices]
    
    # Run KMeans with specific seeds (4 clusters for video game data)
    kmeans = KMeans(n_clusters=4, init=initial_centroids, n_init=1, random_state=42)
    df_temp['Cluster'] = kmeans.fit_predict(X_scaled)
    
    # Create count matrix
    count_matrix = pd.crosstab(df_temp['Rating'], df_temp['Cluster'], margins=True, margins_name='Total')
    
    # Display section
    print(f"\n{'='*80}")
    print(f"RATING-CLUSTER DISTRIBUTION MATRIX FOR SEED SET {i} (Seeds: {seed_indices})")
    print("(Format: Count)")
    print(f"{'='*80}\n")
    
    # Style and display the count matrix
    styled_count = count_matrix.style.set_caption(
        f"Seed Set {i} - Rating-Cluster Distribution"
    ).set_table_styles([
        {'selector': 'caption', 'props': [('font-size', '16px'), ('font-weight', 'bold'), ('text-align', 'center')]},
        {'selector': 'th', 'props': [('text-align', 'center')]},
        {'selector': '.row_heading, .col_heading', 'props': [('font-weight', 'bold')]}
    ])
    display(styled_count)
    
    # Show the count matrix with gradient coloring
    print("\nCount Matrix with Gradient Coloring:")
    styled_counts = count_matrix.style.background_gradient(
        cmap='Blues',
        subset=pd.IndexSlice[count_matrix.index.drop("Total"), count_matrix.columns.drop("Total")]
    )
    display(styled_counts)

import numpy as np
import pandas as pd
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler

# Simulated video game dataset with 8021 data points
np.random.seed(0)
df_clean = pd.DataFrame({
    'Critic_Score': np.random.randint(50, 100, size=8021),

import pandas as pd
import numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# Simulated video game dataset with 8021 data points
np.random.seed(0)
df_clean = pd.DataFrame({
    'Critic_Score': np.random.randint(50, 100, size=8021),
    'User_Score': np.random.uniform(5.0, 10.0, size=8021),
    'Global_Sales': np.random.uniform(0.1, 10.0, size=8021),
    'Rating': np.random.choice(['T', 'E', 'M', 'E10+'], size=8021)
})

# Step 1: Prepare features and labels
features = ['Critic_Score', 'User_Score', 'Global_Sales']
X = df_clean[features]
y_true = df_clean['Rating']

# Step 2: Standardize features
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Step 3: Create consensus matrix
n_runs = 20
n_clusters = 4
n_samples = X_scaled.shape[0]
consensus_matrix = np.zeros((n_samples, n_samples))

for seed in range(n_runs):
    kmeans = KMeans(n_clusters=n_clusters, n_init=1, random_state=seed)
    labels = kmeans.fit_predict(X_scaled)
    for i in range(n_samples):
        for j in range(n_samples):
            if labels[i] == labels[j]:
                consensus_matrix[i, j] += 1

# Step 4: Normalize consensus matrix
consensus_matrix /= n_runs

# Step 5: Apply KMeans to consensus matrix
kmeans_final = KMeans(n_clusters=4, n_init=10, random_state=42)
cluster_labels = kmeans_final.fit_predict(consensus_matrix)

# Step 6: Build summary comparison
summary_df = pd.crosstab(index=y_true, columns=cluster_labels,
                         rownames=["True Label (Rating)"], colnames=["Cluster"])
summary_df = summary_df.reindex(columns=range(4), fill_value=0)

# Step 7: Add totals
summary_df['Total'] = summary_df.sum(axis=1)
summary_df.loc['Total'] = summary_df.sum()

# Step 8: Display result
print("=== True Labels vs Clusters (Video Game Dataset via Consensus Clustering) ===")
print(summary_df)

import pandas as pd
from sklearn.tree import DecisionTreeClassifier, plot_tree
import matplotlib.pyplot as plt
from sklearn.preprocessing import LabelEncoder

# Recreate the consensus matrix as a dataset
data = []

# Consensus matrix counts
matrix = {
    "E":    [518, 513, 494, 508],
    "E10+": [531, 497, 479, 475],
    "M":    [511, 497, 473, 484],
    "T":    [504, 521, 495, 521]
}

# Create rows for each (cluster, class) combination
for cluster in range(4):
    for rating, counts in matrix.items():
        count = counts[cluster]
        data.extend([[cluster, rating]] * count)

# Create DataFrame
df = pd.DataFrame(data, columns=["Cluster", "Rating"])

# Encode rating labels
le = LabelEncoder()
df["Rating_encoded"] = le.fit_transform(df["Rating"])

# Features and labels
X = df[["Cluster"]]
y = df["Rating_encoded"]

# Train Decision Tree with Gini (default)
clf = DecisionTreeClassifier(criterion="gini", max_depth=4, random_state=42)
clf.fit(X, y)

# Plot the decision tree
plt.figure(figsize=(16, 8))
plot_tree(
    clf,
    feature_names=["Cluster"],
    class_names=le.classes_,
    filled=True,
    rounded=True,
    fontsize=10
)
plt.title("Decision Tree Based on Cluster → Rating (Gini, max_depth=4)")
plt.show()
